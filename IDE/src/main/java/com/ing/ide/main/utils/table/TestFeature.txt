import yfinance as yf
import pandas as pd
import ta
import matplotlib.pyplot as plt

# =========================
# DOWNLOAD DATA
# =========================

df = yf.download(
    "BTC-USD",
    start="2020-01-01",
    interval="1d",
    auto_adjust=True
)

# Fix dimensions
close = df["Close"].squeeze()
high = df["High"].squeeze()
low = df["Low"].squeeze()

# =========================
# INDICATORS
# =========================

df["EMA20"] = ta.trend.ema_indicator(close, window=20)
df["EMA200"] = ta.trend.ema_indicator(close, window=200)

df["RSI"] = ta.momentum.rsi(close, window=14)

# =========================
# BACKTEST
# =========================

initial_balance = 100000
balance = initial_balance

position = 0
entry_price = 0
btc_amount = 0

equity_curve = []

trade_count = 0

for i in range(200, len(df)):

    current_close = close.iloc[i]
    prev_rsi = df["RSI"].iloc[i - 1]

    rsi = df["RSI"].iloc[i]

    ema20 = df["EMA20"].iloc[i]
    ema200 = df["EMA200"].iloc[i]

    # =====================
    # BUY
    # =====================

    if position == 0:

        rsi_cross = prev_rsi < 40 and rsi > 40

        if (
            current_close > ema200 and
            current_close > ema20 and
            rsi_cross
        ):

            btc_amount = balance / current_close

            entry_price = current_close

            position = 1

            trade_count += 1

            print(
                f"BUY  {df.index[i].date()} "
                f"@ {current_close:.2f}"
            )

    # =====================
    # SELL
    # =====================

    elif position == 1:

        stop_loss = entry_price * 0.92

        if (
            rsi > 70 or
            current_close < ema20 or
            current_close < stop_loss
        ):

            balance = btc_amount * current_close

            print(
                f"SELL {df.index[i].date()} "
                f"@ {current_close:.2f}"
            )

            position = 0

    # =====================
    # EQUITY CURVE
    # =====================

    current_equity = balance

    if position == 1:
        current_equity = btc_amount * current_close

    equity_curve.append(current_equity)

# =========================
# FINAL RESULTS
# =========================

total_return = (
    (equity_curve[-1] - initial_balance)
    / initial_balance
) * 100

print("\n===== RESULTS =====")
print(f"Initial Balance: {initial_balance:,.2f}")
print(f"Final Balance:   {equity_curve[-1]:,.2f}")
print(f"Total Return:    {total_return:.2f}%")
print(f"Trades Taken:    {trade_count}")

# =========================
# PLOT
# =========================

plt.figure(figsize=(12,6))
plt.plot(equity_curve)

plt.title("BTC Swing Trading Equity Curve")

plt.xlabel("Days")
plt.ylabel("Portfolio Value")

plt.grid(True)

plt.show()
