def timeframe_to_timedelta(timeframe: str) -> pd.Timedelta:
    """
    Supports intraday candles only.
    Daily candles are intentionally not supported here.
    """
    if timeframe.endswith("m"):
        return pd.Timedelta(minutes=int(timeframe[:-1]))
    if timeframe.endswith("h"):
        return pd.Timedelta(hours=int(timeframe[:-1]))

    raise ValueError(
        "This script is intended for intraday timeframes only, such as '15m', '1h', or '4h'. "
        "Daily candles are not used."
    )


def bars_per_year(timeframe: str) -> float:
    td = timeframe_to_timedelta(timeframe)
    seconds_per_bar = td.total_seconds()
    seconds_per_year = 365.25 * 24 * 60 * 60
    return seconds_per_year / seconds_per_bar

def fetch_ohlcv(symbol, timeframe, start_date):
    since = exchange.parse8601(start_date)
    all_candles = []

    print("\n" + "#" * 100)
    print(f"FETCHING DATA FOR {symbol}")
    print("#" * 100)

    while True:
        candles = exchange.fetch_ohlcv(
            symbol,
            timeframe=timeframe,
            since=since,
            limit=1000
        )

        if len(candles) == 0:
            break

        all_candles.extend(candles)

        last_timestamp = candles[-1][0]
        since = last_timestamp + 1

        print(
            f"{symbol}: fetched until "
            f"{datetime.fromtimestamp(last_timestamp / 1000, tz=timezone.utc)}"
        )

        time.sleep(exchange.rateLimit / 1000)

        if len(candles) < 1000:
            break

    if len(all_candles) == 0:
        return pd.DataFrame()

    df = pd.DataFrame(
        all_candles,
        columns=["timestamp", "open", "high", "low", "close", "volume"]
    )

    df["datetime"] = pd.to_datetime(df["timestamp"], unit="ms", utc=True)
    df.set_index("datetime", inplace=True)

    df = df[["open", "high", "low", "close", "volume"]]

    return df


def clean_ohlcv(df, symbol, timeframe):
    if df.empty:
        print(f"{symbol}: no data returned.")
        return df

    df = df.copy()

    df = df[~df.index.duplicated(keep="first")]
    df = df.sort_index()

    df = df.replace([np.inf, -np.inf], np.nan)
    df = df.dropna(subset=["open", "high", "low", "close", "volume"])

    df = df[
        (df["open"] > 0) &
        (df["high"] > 0) &
        (df["low"] > 0) &
        (df["close"] > 0)
    ]

    # Remove possibly incomplete latest candle
    candle_delta = timeframe_to_timedelta(timeframe)
    now_utc = pd.Timestamp.now(tz="UTC")

    if len(df) > 0:
        last_candle_open = df.index[-1]
        if last_candle_open + candle_delta > now_utc:
            print(f"{symbol}: dropping incomplete latest candle at {last_candle_open}")
            df = df.iloc[:-1]

    # Basic missing candle warning
    if len(df) > 1:
        expected_delta = candle_delta
        actual_deltas = df.index.to_series().diff().dropna()
        missing_gaps = actual_deltas[actual_deltas > expected_delta]

        if len(missing_gaps) > 0:
            print(f"{symbol}: warning - found {len(missing_gaps)} possible missing candle gaps.")

    return df

def add_indicators(df):
    df = df.copy()

    close = df["close"]
    high = df["high"]
    low = df["low"]

    df["ema20"] = ta.trend.ema_indicator(close, window=20)
    df["ema50"] = ta.trend.ema_indicator(close, window=50)
    df["ema200"] = ta.trend.ema_indicator(close, window=200)

    df["atr"] = ta.volatility.AverageTrueRange(
        high=high,
        low=low,
        close=close,
        window=14
    ).average_true_range()

    # Previous 20-bar high.
    # Shifted by 1 to prevent current candle high from being used.
    df["hh20"] = high.rolling(BREAKOUT_LOOKBACK).max().shift(1)

    df["entry_signal"] = (
        (df["close"] > df["ema200"]) &
        (df["ema20"] > df["ema50"]) &
        (df["close"] > df["hh20"])
    )

    # Execute signal on the next candle open.
    df["entry_signal_prev"] = df["entry_signal"].shift(1)
    df["atr_prev"] = df["atr"].shift(1)

    df = df.dropna(subset=[
        "ema20",
        "ema50",
        "ema200",
        "atr",
        "hh20",
        "entry_signal_prev",
        "atr_prev"
    ])

    return df

# PORTFOLIO HELPERS
# =====================================================

def portfolio_equity(cash, positions, last_prices):
    equity = cash

    for symbol, pos in positions.items():
        if symbol in last_prices and last_prices[symbol] is not None:
            equity += pos["position_size"] * last_prices[symbol]

    return equity


def max_consecutive_losses(trades_df):
    if trades_df.empty:
        return 0

    trades_df = trades_df.sort_values("exit_time")

    max_streak = 0
    current_streak = 0

    for pnl in trades_df["net_pnl"]:
        if pnl <= 0:
            current_streak += 1
            max_streak = max(max_streak, current_streak)
        else:
            current_streak = 0

    return max_streak


# =====================================================
# PORTFOLIO BACKTEST
# =====================================================

def run_portfolio_backtest(data):
    """
    True multi-symbol portfolio backtest.

    - Single cash balance.
    - Multiple symbols processed chronologically.
    - Entries use previous candle signal and current candle open.
    - Stops are checked intrabar using current candle low.
    - Trend exits use current candle close.
    - Long-only spot-style accounting.
    """

    cash = INITIAL_BALANCE
    positions = {}
    trades = []
    equity_records = []
    last_prices = {}

    all_timestamps = sorted(set().union(*[df.index for df in data.values()]))

    if len(all_timestamps) == 0:
        return trades, pd.DataFrame(), cash

    for current_time in all_timestamps:
        current_rows = {}

        for symbol, df in data.items():
            if current_time in df.index:
                row = df.loc[current_time]
                current_rows[symbol] = row
                last_prices[symbol] = row["close"]

        # =================================================
        # EXIT LOGIC FIRST
        # =================================================

        for symbol in list(positions.keys()):
            if symbol not in current_rows:
                continue

            row = current_rows[symbol]
            pos = positions[symbol]

            exit_signal = False
            exit_reason = None
            raw_exit_price = None

            # Stop loss check.
            # If candle opens below stop, assume worse fill at open.
            if row["low"] <= pos["stop_loss"]:
                exit_signal = True
                exit_reason = "Stop Loss"

                if row["open"] < pos["stop_loss"]:
                    raw_exit_price = row["open"]
                else:
                    raw_exit_price = pos["stop_loss"]

            # Trend exit at candle close.
            elif row["close"] < row["ema20"]:
                exit_signal = True
                exit_reason = "Close Below EMA20"
                raw_exit_price = row["close"]

            if exit_signal:
                # Long exit gets negative slippage.
                exit_price = raw_exit_price * (1 - SLIPPAGE_RATE)

                position_size = pos["position_size"]
                exit_value = position_size * exit_price
                exit_fee = exit_value * FEE_RATE

                cash += exit_value - exit_fee

                gross_pnl = (exit_price - pos["entry_price"]) * position_size
                total_fees = pos["entry_fee"] + exit_fee
                net_pnl = gross_pnl - total_fees

                actual_risk = pos["actual_cash_risk"]
                r_multiple = net_pnl / actual_risk if actual_risk > 0 else np.nan

                equity_after = portfolio_equity(cash, positions, last_prices)

                trade = {
                    "symbol": symbol,
                    "entry_time": pos["entry_time"],
                    "exit_time": current_time,
                    "entry_price": pos["entry_price"],
                    "exit_price": exit_price,
                    "stop_loss": pos["stop_loss"],
                    "position_size": position_size,
                    "entry_value": pos["entry_value"],
                    "exit_value": exit_value,
                    "gross_pnl": gross_pnl,
                    "entry_fee": pos["entry_fee"],
                    "exit_fee": exit_fee,
                    "fees": total_fees,
                    "net_pnl": net_pnl,
                    "actual_cash_risk": actual_risk,
                    "r_multiple": r_multiple,
                    "exit_reason": exit_reason,
                    "holding_period": current_time - pos["entry_time"],
                    "cash_after": cash,
                    "equity_after": equity_after
                }

                trades.append(trade)
                del positions[symbol]

        # =================================================
        # ENTRY LOGIC SECOND
        # =================================================

        for symbol, row in current_rows.items():
            if symbol in positions:
                continue

            if len(positions) >= MAX_OPEN_POSITIONS:
                continue

            if row["entry_signal_prev"] != True:
                continue

            if pd.isna(row["atr_prev"]) or row["atr_prev"] <= 0:
                continue

            current_equity = portfolio_equity(cash, positions, last_prices)

            if current_equity <= 0 or cash <= 0:
                continue

            # Enter at next candle open with positive slippage.
            entry_price = row["open"] * (1 + SLIPPAGE_RATE)

            stop_loss = entry_price - (ATR_MULTIPLIER * row["atr_prev"])

            if stop_loss <= 0:
                continue

            risk_per_unit = entry_price - stop_loss

            if risk_per_unit <= 0:
                continue

            requested_risk = RISK_PER_TRADE_USD

            # Risk-based size.
            size_by_risk = requested_risk / risk_per_unit
            desired_notional = size_by_risk * entry_price

            # Capital constraints.
            max_notional_by_cash = cash / (1 + FEE_RATE)
            max_notional_by_exposure = current_equity * MAX_POSITION_VALUE_PCT

            max_allowed_notional = min(max_notional_by_cash, max_notional_by_exposure)

            entry_value = min(desired_notional, max_allowed_notional)

            if entry_value < MIN_NOTIONAL:
                continue

            position_size = entry_value / entry_price
            entry_fee = entry_value * FEE_RATE

            total_cost = entry_value + entry_fee

            if total_cost > cash:
                continue

            actual_cash_risk = risk_per_unit * position_size

            cash -= total_cost

            positions[symbol] = {
                "symbol": symbol,
                "entry_time": current_time,
                "entry_price": entry_price,
                "stop_loss": stop_loss,
                "position_size": position_size,
                "entry_value": entry_value,
                "entry_fee": entry_fee,
                "actual_cash_risk": actual_cash_risk
            }

        # =================================================
        # EQUITY RECORD
        # =================================================

        equity = portfolio_equity(cash, positions, last_prices)

        equity_records.append({
            "datetime": current_time,
            "cash": cash,
            "equity": equity,
            "open_positions": len(positions)
        })

    # =====================================================
    # FORCE CLOSE OPEN POSITIONS AT END
    # =====================================================

    if len(positions) > 0:
        print("\nForcing open positions closed at final available close...")

    for symbol in list(positions.keys()):
        pos = positions[symbol]
        df = data[symbol]

        final_time = df.index[-1]
        final_close = df["close"].iloc[-1]
        exit_price = final_close * (1 - SLIPPAGE_RATE)

        position_size = pos["position_size"]
        exit_value = position_size * exit_price
        exit_fee = exit_value * FEE_RATE

        cash += exit_value - exit_fee

        gross_pnl = (exit_price - pos["entry_price"]) * position_size
        total_fees = pos["entry_fee"] + exit_fee
        net_pnl = gross_pnl - total_fees

        actual_risk = pos["actual_cash_risk"]
        r_multiple = net_pnl / actual_risk if actual_risk > 0 else np.nan

        trade = {
            "symbol": symbol,
            "entry_time": pos["entry_time"],
            "exit_time": final_time,
            "entry_price": pos["entry_price"],
            "exit_price": exit_price,
            "stop_loss": pos["stop_loss"],
            "position_size": position_size,
            "entry_value": pos["entry_value"],
            "exit_value": exit_value,
            "gross_pnl": gross_pnl,
            "entry_fee": pos["entry_fee"],
            "exit_fee": exit_fee,
            "fees": total_fees,
            "net_pnl": net_pnl,
            "actual_cash_risk": actual_risk,
            "r_multiple": r_multiple,
            "exit_reason": "End Of Backtest",
            "holding_period": final_time - pos["entry_time"],
            "cash_after": cash,
            "equity_after": cash
        }

        trades.append(trade)
        del positions[symbol]

    final_timestamp = max([df.index[-1] for df in data.values()])
    equity_records.append({
        "datetime": final_timestamp,
        "cash": cash,
        "equity": cash,
        "open_positions": 0
    })

    equity_curve = pd.DataFrame(equity_records)
    equity_curve = equity_curve.drop_duplicates(subset=["datetime"], keep="last")
    equity_curve = equity_curve.sort_values("datetime")
    equity_curve.set_index("datetime", inplace=True)

    return trades, equity_curve, cash

# METRICS
# =====================================================

def calculate_metrics(trades, equity_curve):
    metrics = {}

    if equity_curve.empty:
        return metrics

    start_equity = INITIAL_BALANCE
    final_equity = equity_curve["equity"].iloc[-1]

    total_return = (final_equity / start_equity) - 1

    running_max = equity_curve["equity"].cummax()
    drawdown = (equity_curve["equity"] / running_max) - 1
    max_drawdown = drawdown.min()

    returns = equity_curve["equity"].pct_change().replace([np.inf, -np.inf], np.nan).dropna()

    annualization = bars_per_year(TIMEFRAME)

    if len(returns) > 1 and returns.std() > 0:
        sharpe = (returns.mean() / returns.std()) * math.sqrt(annualization)
    else:
        sharpe = np.nan

    downside_returns = returns[returns < 0]

    if len(downside_returns) > 1 and downside_returns.std() > 0:
        sortino = (returns.mean() / downside_returns.std()) * math.sqrt(annualization)
    else:
        sortino = np.nan

    start_date = equity_curve.index[0]
    end_date = equity_curve.index[-1]
    test_days = (end_date - start_date).days

    if test_days > 0 and final_equity > 0:
        cagr = (final_equity / start_equity) ** (365.25 / test_days) - 1
    else:
        cagr = np.nan

    calmar = cagr / abs(max_drawdown) if max_drawdown < 0 and not pd.isna(cagr) else np.nan

    exposure = equity_curve["open_positions"].gt(0).mean()

    metrics.update({
        "initial_balance": start_equity,
        "final_equity": final_equity,
        "total_return_pct": total_return * 100,
        "cagr_pct": cagr * 100 if not pd.isna(cagr) else np.nan,
        "max_drawdown_pct": max_drawdown * 100,
        "sharpe_4h": sharpe,
        "sortino_4h": sortino,
        "calmar": calmar,
        "exposure_pct": exposure * 100,
        "test_start": start_date,
        "test_end": end_date,
        "test_days": test_days
    })

    if len(trades) == 0:
        metrics.update({
            "total_trades": 0
        })
        return metrics

    trades_df = pd.DataFrame(trades)

    wins = trades_df[trades_df["net_pnl"] > 0]
    losses = trades_df[trades_df["net_pnl"] <= 0]

    gross_profit = wins["net_pnl"].sum()
    gross_loss = abs(losses["net_pnl"].sum())

    profit_factor = gross_profit / gross_loss if gross_loss > 0 else np.inf

    avg_win = wins["net_pnl"].mean() if len(wins) > 0 else 0
    avg_loss = losses["net_pnl"].mean() if len(losses) > 0 else 0

    win_rate = len(wins) / len(trades_df)

    expectancy = trades_df["net_pnl"].mean()

    avg_r = trades_df["r_multiple"].mean()
    median_r = trades_df["r_multiple"].median()

    payoff_ratio = avg_win / abs(avg_loss) if avg_loss < 0 else np.inf

    max_loss_streak = max_consecutive_losses(trades_df)

    metrics.update({
        "total_trades": len(trades_df),
        "wins": len(wins),
        "losses": len(losses),
        "win_rate_pct": win_rate * 100,
        "gross_profit": gross_profit,
        "gross_loss": gross_loss,
        "total_net_pnl": trades_df["net_pnl"].sum(),
        "avg_win": avg_win,
        "avg_loss": avg_loss,
        "expectancy": expectancy,
        "profit_factor": profit_factor,
        "payoff_ratio": payoff_ratio,
        "avg_r": avg_r,
        "median_r": median_r,
        "best_trade": trades_df["net_pnl"].max(),
        "worst_trade": trades_df["net_pnl"].min(),
        "max_consecutive_losses": max_loss_streak,
        "avg_holding_period": trades_df["holding_period"].mean()
    })

    return metrics

def calculate_equal_weight_buy_hold(data):
    """
    Simple equal-weight buy-and-hold benchmark using the same intraday data.
    No daily candles and no daily resampling.
    """

    if len(data) == 0:
        return None

    common_start = max(df.index.min() for df in data.values())
    common_end = min(df.index.max() for df in data.values())

    if common_start >= common_end:
        return None

    allocation = INITIAL_BALANCE / len(data)
    final_value = 0.0

    details = []

    for symbol, df in data.items():
        first_row = df[df.index >= common_start].iloc[0]
        last_row = df[df.index <= common_end].iloc[-1]

        entry_price = first_row["close"] * (1 + SLIPPAGE_RATE)
        exit_price = last_row["close"] * (1 - SLIPPAGE_RATE)

        # Allocation includes entry fee.
        quantity = allocation / (entry_price * (1 + FEE_RATE))

        exit_value = quantity * exit_price
        exit_fee = exit_value * FEE_RATE
        final_symbol_value = exit_value - exit_fee

        final_value += final_symbol_value

        details.append({
            "symbol": symbol,
            "entry_price": entry_price,
            "exit_price": exit_price,
            "quantity": quantity,
            "start_value": allocation,
            "final_value": final_symbol_value,
            "return_pct": ((final_symbol_value / allocation) - 1) * 100
        })

    return {
        "start": common_start,
        "end": common_end,
        "initial_balance": INITIAL_BALANCE,
        "final_value": final_value,
        "return_pct": ((final_value / INITIAL_BALANCE) - 1) * 100,
        "details": details
    }

def main():
    data = {}

    for symbol in SYMBOLS:
        df = fetch_ohlcv(symbol, TIMEFRAME, START_DATE)
        df = clean_ohlcv(df, symbol, TIMEFRAME)

        if df.empty:
            print(f"{symbol}: skipped because no usable data was found.")
            continue

        df = add_indicators(df)

        if df.empty:
            print(f"{symbol}: skipped because not enough data for indicators.")
            continue

        data[symbol] = df

        print(f"{symbol}: ready with {len(df)} candles from {df.index[0]} to {df.index[-1]}")

    if len(data) == 0:
        print("No usable data. Backtest stopped.")
        return

    trades, equity_curve, final_cash = run_portfolio_backtest(data)

    metrics = calculate_metrics(trades, equity_curve)
    benchmark = calculate_equal_weight_buy_hold(data)

    if PRINT_TRADES:
        print_trade_history(trades)

    print_summary(metrics)
    print_benchmark(benchmark)

    if EXPORT_CSV:
        trades_df = pd.DataFrame(trades)

        if not trades_df.empty:
            trades_df.to_csv("trades.csv", index=False)
            print("\nSaved trades to trades.csv")

        if not equity_curve.empty:
            equity_curve.to_csv("equity_curve.csv")
            print("Saved equity curve to equity_curve.csv")


if __name__ == "__main__":
    main()
