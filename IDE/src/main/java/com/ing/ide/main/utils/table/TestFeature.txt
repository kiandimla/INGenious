import yfinance as yf
import pandas as pd
import numpy as np
import ta
import matplotlib.pyplot as plt

# =========================
# Download BTC data
# =========================

df = yf.download(
    "BTC-USD",
    start="2020-01-01",
    interval="1d",
    auto_adjust=True
)

# Convert to proper Series
close = df["Close"].squeeze()
high = df["High"].squeeze()
low = df["Low"].squeeze()

# =========================
# Indicators
# =========================

df["EMA200"] = ta.trend.ema_indicator(close, window=200)

df["RSI"] = ta.momentum.rsi(close, window=14)

macd = ta.trend.MACD(close)

df["MACD_HIST"] = macd.macd_diff()

atr = ta.volatility.AverageTrueRange(
    high,
    low,
    close,
    window=14
)

df["ATR"] = atr.average_true_range()

# =========================
# Backtest
# =========================

initial_balance = 100000
balance = initial_balance

position = 0
entry_price = 0
btc_amount = 0

risk_per_trade = 0.02

equity_curve = []

for i in range(200, len(df)):

    current_close = close.iloc[i]
    ema200 = df["EMA200"].iloc[i]
    rsi = df["RSI"].iloc[i]
    macd_hist = df["MACD_HIST"].iloc[i]
    current_atr = df["ATR"].iloc[i]

    # =====================
    # ENTRY
    # =====================

    if position == 0:

        if (
            current_close > ema200 and
            rsi < 35 and
            macd_hist > 0
        ):

            stop_loss = current_close - (2 * current_atr)

            risk_amount = balance * risk_per_trade

            position_size = risk_amount / (
                current_close - stop_loss
            )

            btc_amount = position_size

            entry_price = current_close

            position = 1

            print(f"BUY  {df.index[i].date()} @ {current_close:.2f}")

    # =====================
    # EXIT
    # =====================

    elif position == 1:

        stop_loss = entry_price - (2 * current_atr)

        if (
            rsi > 65 or
            current_close < stop_loss or
            current_close < ema200
        ):

            pnl = (
                current_close - entry_price
            ) * btc_amount

            balance += pnl

            print(
                f"SELL {df.index[i].date()} @ "
                f"{current_close:.2f} | "
                f"PnL: {pnl:.2f}"
            )

            position = 0

    current_equity = balance

    if position == 1:
        unrealized = (
            current_close - entry_price
        ) * btc_amount

        current_equity += unrealized

    equity_curve.append(current_equity)

# =========================
# Results
# =========================

total_return = (
    (equity_curve[-1] - initial_balance)
    / initial_balance
) * 100

print("\n===== RESULTS =====")
print(f"Initial: {initial_balance:,.2f}")
print(f"Final:   {equity_curve[-1]:,.2f}")
print(f"Return:  {total_return:.2f}%")

# =========================
# Plot
# =========================

plt.figure(figsize=(12,6))
plt.plot(equity_curve)
plt.title("BTC Swing Strategy Equity Curve")
plt.xlabel("Time")
plt.ylabel("Balance")
plt.grid(True)
plt.show()
