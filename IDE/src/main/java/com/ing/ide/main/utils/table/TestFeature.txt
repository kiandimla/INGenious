import ccxt
import pandas as pd
import numpy as np
import ta
from datetime import datetime, timezone
import time

# =========================================================
# STRATEGY SUMMARY
# =========================================================
#
# MULTI-TIMEFRAME TREND FOLLOWING STRATEGY
#
# HIGHER TIMEFRAME (4H)
# ---------------------
# Determines overall bullish trend direction.
#
# Conditions:
# - Price above EMA200
# - EMA20 above EMA50
#
# MID TIMEFRAME (1H)
# ------------------
# Confirms momentum alignment.
#
# Conditions:
# - Price above EMA200
# - RSI > 50
#
# ENTRY TIMEFRAME (15M)
# ---------------------
# Finds breakout entry opportunity.
#
# Conditions:
# - Close above previous 20 candle high
# - RSI > 50
#
# ENTRY
# -----
# - Enter next candle open
#
# STOP LOSS
# ----------
# - 0.5 ATR below entry
#
# TAKE PROFIT
# -----------
# - Fixed 2:1 Reward-To-Risk
#
# RISK
# ----
# - Fixed $10 risk per trade
#
# REALISM
# -------
# - Includes fees
# - Includes slippage
#
# =========================================================


# =========================================================
# CONFIG
# =========================================================

# Use ONE symbol first while debugging.
# Add ETH/SOL later once everything works.

SYMBOLS = ["BTC/USDT"]

ENTRY_TIMEFRAME = "15m"
MID_TIMEFRAME = "1h"
HTF_TIMEFRAME = "4h"

# Keep the history smaller first for speed/debugging.
START_DATE = "2025-01-01T00:00:00Z"

RISK_PER_TRADE = 10
RR_RATIO = 2.0

FEE_RATE = 0.001
SLIPPAGE_RATE = 0.0005

BREAKOUT_LOOKBACK = 20

# Limit candle holding duration for performance.
MAX_HOLD_CANDLES = 50

exchange = ccxt.binance({
    "enableRateLimit": True
})


# =========================================================
# DATA FETCHING
# =========================================================

def fetch_ohlcv(symbol, timeframe, start_date):

    since = exchange.parse8601(start_date)

    all_candles = []

    loops = 0
    max_loops = 50

    while loops < max_loops:

        loops += 1

        candles = exchange.fetch_ohlcv(
            symbol,
            timeframe=timeframe,
            since=since,
            limit=1000
        )

        if len(candles) == 0:
            break

        all_candles.extend(candles)

        last_timestamp = candles[-1][0]

        since = last_timestamp + 1

        print(
            f"{symbol} {timeframe}: fetched until "
            f"{datetime.fromtimestamp(last_timestamp / 1000, tz=timezone.utc)}"
        )

        time.sleep(exchange.rateLimit / 1000)

        if len(candles) < 1000:
            break

    df = pd.DataFrame(
        all_candles,
        columns=[
            "timestamp",
            "open",
            "high",
            "low",
            "close",
            "volume"
        ]
    )

    df["datetime"] = pd.to_datetime(
        df["timestamp"],
        unit="ms",
        utc=True
    )

    df.set_index("datetime", inplace=True)

    df = df[[
        "open",
        "high",
        "low",
        "close",
        "volume"
    ]]

    print(f"{symbol} {timeframe}: FETCH COMPLETE")
    print(f"Total candles fetched: {len(df)}")

    return df


# =========================================================
# INDICATORS
# =========================================================

def add_indicators(df):

    df = df.copy()

    close = df["close"]
    high = df["high"]
    low = df["low"]

    # Trend EMAs
    df["ema20"] = ta.trend.ema_indicator(close, window=20)
    df["ema50"] = ta.trend.ema_indicator(close, window=50)
    df["ema200"] = ta.trend.ema_indicator(close, window=200)

    # RSI
    df["rsi"] = ta.momentum.RSIIndicator(
        close,
        window=14
    ).rsi()

    # ATR
    df["atr"] = ta.volatility.AverageTrueRange(
        high=high,
        low=low,
        close=close,
        window=14
    ).average_true_range()

    # Previous 20-candle high
    df["hh20"] = (
        high
        .rolling(BREAKOUT_LOOKBACK)
        .max()
        .shift(1)
    )

    return df


# =========================================================
# HIGHER TIMEFRAME BIAS
# =========================================================

def htf_bullish(row):

    return (
        row["close"] > row["ema200"] and
        row["ema20"] > row["ema50"]
    )


# =========================================================
# MID TIMEFRAME CONFIRMATION
# =========================================================

def mtf_confirmation(row):

    return (
        row["close"] > row["ema200"] and
        row["rsi"] > 50
    )


# =========================================================
# ENTRY SIGNAL
# =========================================================

def entry_signal(row):

    return (
        row["close"] > row["hh20"] and
        row["rsi"] > 50
    )


# =========================================================
# BACKTEST
# =========================================================

def run_backtest(symbol):

    print("\n" + "=" * 80)
    print(f"RUNNING BACKTEST FOR {symbol}")
    print("=" * 80)

    # =====================================================
    # FETCH DATA
    # =====================================================

    df_15m = add_indicators(
        fetch_ohlcv(
            symbol,
            ENTRY_TIMEFRAME,
            START_DATE
        )
    )

    df_1h = add_indicators(
        fetch_ohlcv(
            symbol,
            MID_TIMEFRAME,
            START_DATE
        )
    )

    df_4h = add_indicators(
        fetch_ohlcv(
            symbol,
            HTF_TIMEFRAME,
            START_DATE
        )
    )

    # Remove NaNs from indicators
    df_15m.dropna(inplace=True)
    df_1h.dropna(inplace=True)
    df_4h.dropna(inplace=True)

    trades = []

    trade_open = False

    htf_pass = 0
    mtf_pass = 0
    entry_pass = 0

    # =====================================================
    # MAIN LOOP
    # =====================================================

    for i in range(1, len(df_15m) - 1):

        if i % 500 == 0:
            print(f"Processing candle {i}/{len(df_15m)}")

        # Prevent overlapping trades
        if trade_open:
            continue

        current_time = df_15m.index[i]

        row_15m = df_15m.iloc[i]

        # =================================================
        # GET LATEST AVAILABLE 1H / 4H CANDLE
        # =================================================

        df_1h_filtered = df_1h[
            df_1h.index <= current_time
        ]

        df_4h_filtered = df_4h[
            df_4h.index <= current_time
        ]

        if (
            len(df_1h_filtered) == 0 or
            len(df_4h_filtered) == 0
        ):
            continue

        row_1h = df_1h_filtered.iloc[-1]
        row_4h = df_4h_filtered.iloc[-1]

        # =================================================
        # HIGHER TIMEFRAME BIAS
        # =================================================

        if not htf_bullish(row_4h):
            continue

        htf_pass += 1

        # =================================================
        # MID TIMEFRAME CONFIRMATION
        # =================================================

        if not mtf_confirmation(row_1h):
            continue

        mtf_pass += 1

        # =================================================
        # ENTRY SIGNAL
        # =================================================

        if not entry_signal(row_15m):
            continue

        entry_pass += 1

        # =================================================
        # ENTRY EXECUTION
        # =================================================

        next_row = df_15m.iloc[i + 1]

        entry_time = next_row.name

        # Realistic slippage
        entry_price = (
            next_row["open"]
            * (1 + SLIPPAGE_RATE)
        )

        # Smaller ATR stop for more realistic crypto entries
        atr = row_15m["atr"] * 0.5

        stop_loss = entry_price - atr

        risk_per_unit = entry_price - stop_loss

        if risk_per_unit <= 0:
            continue

        # Fixed 2:1 reward:risk
        take_profit = (
            entry_price +
            (risk_per_unit * RR_RATIO)
        )

        # Position sizing
        position_size = (
            RISK_PER_TRADE /
            risk_per_unit
        )

        entry_value = (
            position_size *
            entry_price
        )

        entry_fee = (
            entry_value *
            FEE_RATE
        )

        print(
            f"\nTRADE OPENED | "
            f"{symbol} | "
            f"Entry: {entry_price:.2f} | "
            f"SL: {stop_loss:.2f} | "
            f"TP: {take_profit:.2f}"
        )

        trade_open = True

        # =================================================
        # TRADE MANAGEMENT
        # =================================================

        trade_closed = False

        future_limit = min(
            i + MAX_HOLD_CANDLES,
            len(df_15m)
        )

        for j in range(i + 1, future_limit):

            future_row = df_15m.iloc[j]

            high = future_row["high"]
            low = future_row["low"]

            exit_reason = None
            exit_price = None

            # STOP LOSS
            if low <= stop_loss:

                exit_reason = "STOP LOSS"

                exit_price = (
                    stop_loss *
                    (1 - SLIPPAGE_RATE)
                )

            # TAKE PROFIT
            elif high >= take_profit:

                exit_reason = "TAKE PROFIT"

                exit_price = (
                    take_profit *
                    (1 - SLIPPAGE_RATE)
                )

            if exit_reason:

                exit_time = future_row.name

                exit_value = (
                    position_size *
                    exit_price
                )

                exit_fee = (
                    exit_value *
                    FEE_RATE
                )

                gross_pnl = (
                    (exit_price - entry_price)
                    * position_size
                )

                total_fees = (
                    entry_fee +
                    exit_fee
                )

                net_pnl = (
                    gross_pnl -
                    total_fees
                )

                r_multiple = (
                    net_pnl /
                    RISK_PER_TRADE
                )

                holding_period = (
                    exit_time -
                    entry_time
                )

                trade = {
                    "symbol": symbol,
                    "entry_time": entry_time,
                    "exit_time": exit_time,
                    "entry_price": entry_price,
                    "exit_price": exit_price,
                    "stop_loss": stop_loss,
                    "take_profit": take_profit,
                    "position_size": position_size,
                    "gross_pnl": gross_pnl,
                    "fees": total_fees,
                    "net_pnl": net_pnl,
                    "r_multiple": r_multiple,
                    "holding_period": holding_period,
                    "exit_reason": exit_reason
                }

                trades.append(trade)

                print(
                    f"TRADE CLOSED | "
                    f"{exit_reason} | "
                    f"PnL: ${net_pnl:.2f}"
                )

                trade_open = False
                trade_closed = True

                break

        if not trade_closed:
            trade_open = False

    print("\nDEBUG COUNTERS")
    print(f"HTF PASSED: {htf_pass}")
    print(f"MTF PASSED: {mtf_pass}")
    print(f"ENTRY PASSED: {entry_pass}")

    print(f"\n{symbol}: TOTAL TRADES FOUND = {len(trades)}")

    return trades


# =========================================================
# METRICS
# =========================================================

def calculate_metrics(trades):

    if len(trades) == 0:
        return {}

    df = pd.DataFrame(trades)

    wins = df[df["net_pnl"] > 0]
    losses = df[df["net_pnl"] <= 0]

    total_trades = len(df)

    win_rate = (
        len(wins) / total_trades
    ) * 100

    gross_profit = wins["net_pnl"].sum()

    gross_loss = abs(
        losses["net_pnl"].sum()
    )

    profit_factor = (
        gross_profit / gross_loss
        if gross_loss != 0
        else np.inf
    )

    expectancy = df["net_pnl"].mean()

    avg_win = (
        wins["net_pnl"].mean()
        if len(wins) > 0 else 0
    )

    avg_loss = (
        losses["net_pnl"].mean()
        if len(losses) > 0 else 0
    )

    return {
        "total_trades": total_trades,
        "wins": len(wins),
        "losses": len(losses),
        "win_rate": win_rate,
        "gross_profit": gross_profit,
        "gross_loss": gross_loss,
        "net_pnl": df["net_pnl"].sum(),
        "avg_win": avg_win,
        "avg_loss": avg_loss,
        "expectancy": expectancy,
        "profit_factor": profit_factor,
        "best_trade": df["net_pnl"].max(),
        "worst_trade": df["net_pnl"].min(),
        "avg_r": df["r_multiple"].mean()
    }


# =========================================================
# PRINT TRADES
# =========================================================

def print_trades(trades):

    print("\n" + "=" * 120)
    print("TRADE HISTORY")
    print("=" * 120)

    if len(trades) == 0:
        print("No trades found.")
        return

    for idx, trade in enumerate(trades, start=1):

        print(f"\nTRADE #{idx}")
        print("-" * 80)

        print(f"Symbol:          {trade['symbol']}")
        print(f"Entry Time:      {trade['entry_time']}")
        print(f"Exit Time:       {trade['exit_time']}")
        print(f"Holding Period:  {trade['holding_period']}")

        print(f"\nEntry Price:     {trade['entry_price']:.4f}")
        print(f"Stop Loss:       {trade['stop_loss']:.4f}")
        print(f"Take Profit:     {trade['take_profit']:.4f}")
        print(f"Exit Price:      {trade['exit_price']:.4f}")

        print(f"\nPosition Size:   {trade['position_size']:.6f}")

        print(f"\nGross PnL:       ${trade['gross_pnl']:.2f}")
        print(f"Fees:            ${trade['fees']:.2f}")
        print(f"Net PnL:         ${trade['net_pnl']:.2f}")

        print(f"R Multiple:      {trade['r_multiple']:.2f}R")
        print(f"Exit Reason:     {trade['exit_reason']}")


# =========================================================
# PRINT SUMMARY
# =========================================================

def print_summary(metrics):

    print("\n" + "=" * 80)
    print("BACKTEST SUMMARY")
    print("=" * 80)

    if not metrics:
        print("No completed trades.")
        return

    print(f"Total Trades:    {metrics['total_trades']}")
    print(f"Wins:            {metrics['wins']}")
    print(f"Losses:          {metrics['losses']}")
    print(f"Win Rate:        {metrics['win_rate']:.2f}%")

    print(f"\nNet PnL:         ${metrics['net_pnl']:.2f}")
    print(f"Gross Profit:    ${metrics['gross_profit']:.2f}")
    print(f"Gross Loss:      ${metrics['gross_loss']:.2f}")

    print(f"\nAverage Win:     ${metrics['avg_win']:.2f}")
    print(f"Average Loss:    ${metrics['avg_loss']:.2f}")

    print(f"Expectancy:      ${metrics['expectancy']:.2f}")
    print(f"Profit Factor:   {metrics['profit_factor']:.2f}")

    print(f"\nAverage R:       {metrics['avg_r']:.2f}R")
    print(f"Best Trade:      ${metrics['best_trade']:.2f}")
    print(f"Worst Trade:     ${metrics['worst_trade']:.2f}")


# =========================================================
# MAIN
# =========================================================

all_trades = []

for symbol in SYMBOLS:

    trades = run_backtest(symbol)

    print_trades(trades)

    all_trades.extend(trades)

portfolio_metrics = calculate_metrics(all_trades)

print_summary(portfolio_metrics)
