import yfinance as yf
import pandas as pd
import numpy as np
import ta
import matplotlib.pyplot as plt

# =====================================
# DOWNLOAD DATA
# =====================================

df = yf.download(
    "BTC-USD",
    start="2020-01-01",
    interval="1d",
    auto_adjust=True
)

# =====================================
# INDICATORS
# =====================================

df["EMA200"] = ta.trend.ema_indicator(df["Close"], window=200)

df["RSI"] = ta.momentum.rsi(df["Close"], window=14)

macd = ta.trend.MACD(df["Close"])
df["MACD"] = macd.macd()
df["MACD_SIGNAL"] = macd.macd_signal()
df["MACD_HIST"] = macd.macd_diff()

atr = ta.volatility.AverageTrueRange(
    df["High"],
    df["Low"],
    df["Close"],
    window=14
)

df["ATR"] = atr.average_true_range()

# =====================================
# STRATEGY
# =====================================

initial_balance = 100000  # PHP equivalent test capital
balance = initial_balance

position = 0
entry_price = 0
btc_amount = 0

risk_per_trade = 0.02

equity_curve = []

trades = []

for i in range(200, len(df)):

    close = df["Close"].iloc[i]
    ema200 = df["EMA200"].iloc[i]
    rsi = df["RSI"].iloc[i]
    macd_hist = df["MACD_HIST"].iloc[i]
    atr = df["ATR"].iloc[i]

    # ==============================
    # ENTRY
    # ==============================

    if position == 0:

        if (
            close > ema200 and
            rsi < 35 and
            macd_hist > 0
        ):

            stop_loss = close - (2 * atr)

            risk_amount = balance * risk_per_trade

            position_size = risk_amount / (close - stop_loss)

            btc_amount = position_size

            entry_price = close

            position = 1

            trades.append(
                f"BUY {df.index[i].date()} @ {close:.2f}"
            )

    # ==============================
    # EXIT
    # ==============================

    elif position == 1:

        stop_loss = entry_price - (2 * atr)

        take_profit_condition = rsi > 65

        stop_condition = close < stop_loss

        trend_break = close < ema200

        if (
            take_profit_condition or
            stop_condition or
            trend_break
        ):

            pnl = (close - entry_price) * btc_amount

            balance += pnl

            trades.append(
                f"SELL {df.index[i].date()} @ {close:.2f} | PnL: {pnl:.2f}"
            )

            position = 0

    # Equity curve
    current_equity = balance

    if position == 1:
        unrealized = (close - entry_price) * btc_amount
        current_equity += unrealized

    equity_curve.append(current_equity)

# =====================================
# RESULTS
# =====================================

total_return = (
    (equity_curve[-1] - initial_balance)
    / initial_balance
) * 100

print("\n===== RESULTS =====")
print(f"Initial Balance: {initial_balance:,.2f}")
print(f"Final Balance:   {equity_curve[-1]:,.2f}")
print(f"Total Return:    {total_return:.2f}%")
print(f"Trades Taken:    {len(trades)}")

# =====================================
# PLOT
# =====================================

plt.figure(figsize=(12, 6))
plt.plot(equity_curve)
plt.title("Equity Curve")
plt.xlabel("Days")
plt.ylabel("Balance")
plt.grid()
plt.show()
