import yfinance as yf
import pandas as pd
import ta
import matplotlib.pyplot as plt

# =============================
# DOWNLOAD BTC DATA
# =============================

df = yf.download(
    "BTC-USD",
    start="2020-01-01",
    interval="1d",
    auto_adjust=True
)

close = df["Close"].squeeze()
high = df["High"].squeeze()

# =============================
# INDICATORS
# =============================

df["EMA20"] = ta.trend.ema_indicator(close, window=20)
df["EMA50"] = ta.trend.ema_indicator(close, window=50)
df["EMA200"] = ta.trend.ema_indicator(close, window=200)

# Previous 20-day breakout
df["HH20"] = high.rolling(20).max().shift(1)

# =============================
# BACKTEST
# =============================

initial_balance = 100000
balance = initial_balance

position = 0
btc_amount = 0
entry_price = 0

equity_curve = []
trade_count = 0

for i in range(200, len(df)):

    current_close = close.iloc[i]

    ema20 = df["EMA20"].iloc[i]
    ema50 = df["EMA50"].iloc[i]
    ema200 = df["EMA200"].iloc[i]

    hh20 = df["HH20"].iloc[i]

    # =========================
    # BUY
    # =========================

    if position == 0:

        if (
            current_close > ema200 and
            ema20 > ema50 and
            current_close > hh20
        ):

            btc_amount = balance / current_close

            entry_price = current_close

            position = 1
            trade_count += 1

            print(
                f"BUY  {df.index[i].date()} "
                f"@ {current_close:.2f}"
            )

    # =========================
    # SELL
    # =========================

    elif position == 1:

        if current_close < ema20:

            balance = btc_amount * current_close

            print(
                f"SELL {df.index[i].date()} "
                f"@ {current_close:.2f}"
            )

            position = 0

    # =========================
    # EQUITY
    # =========================

    if position == 1:
        equity = btc_amount * current_close
    else:
        equity = balance

    equity_curve.append(equity)

# =============================
# RESULTS
# =============================

final_balance = equity_curve[-1]

total_return = (
    (final_balance - initial_balance)
    / initial_balance
) * 100

print("\n===== RESULTS =====")

print(f"Initial Balance: {initial_balance:,.2f}")
print(f"Final Balance:   {final_balance:,.2f}")

print(f"Total Return:    {total_return:.2f}%")
print(f"Trades Taken:    {trade_count}")

# =============================
# PLOT
# =============================

plt.figure(figsize=(12,6))

plt.plot(equity_curve)

plt.title("BTC Trend Breakout Strategy")

plt.xlabel("Days")
plt.ylabel("Portfolio Value")

plt.grid(True)

plt.show()
