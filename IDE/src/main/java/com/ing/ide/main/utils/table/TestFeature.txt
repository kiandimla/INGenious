import ccxt
import pandas as pd
import numpy as np
import ta
from datetime import datetime, timezone
import time

# =========================================================
# STRATEGY SUMMARY
# =========================================================
#
# STRATEGY TYPE:
# Multi-timeframe discretionary-style trend-following setup
#
# HIGHER TIMEFRAME (4H):
# - Used to determine directional bias
# - Only LONG trades are allowed
# - Price must be above EMA200
# - EMA20 must be above EMA50
#
# MID TIMEFRAME (1H):
# - Used for confirmation
# - Price must also be above EMA200
# - EMA20 must be above EMA50
# - RSI > 50 for bullish momentum
# - MACD histogram > 0 for bullish momentum
#
# ENTRY TIMEFRAME (15M):
# - Entry happens when:
#     - Price closes above previous 20-candle high
#     - RSI > 50
#     - MACD histogram > 0
#
# ENTRY:
# - Enter on NEXT candle open (more realistic)
#
# STOP LOSS:
# - 1 ATR below entry
#
# TAKE PROFIT:
# - Fixed 2:1 Reward:Risk
#
# RISK:
# - Fixed $10 risk per trade
#
# FEES:
# - 0.1% fee per side
#
# SLIPPAGE:
# - 0.05% slippage assumed
#
# =========================================================


# =========================================================
# CONFIG
# =========================================================

SYMBOLS = ["BTC/USDT", "ETH/USDT", "SOL/USDT"]

ENTRY_TIMEFRAME = "15m"
MID_TIMEFRAME = "1h"
HTF_TIMEFRAME = "4h"

START_DATE = "2022-01-01T00:00:00Z"

RISK_PER_TRADE = 10
RR_RATIO = 2.0

FEE_RATE = 0.001
SLIPPAGE_RATE = 0.0005

BREAKOUT_LOOKBACK = 20

exchange = ccxt.binance({
    "enableRateLimit": True
})


# =========================================================
# DATA FETCHING
# =========================================================

def fetch_ohlcv(symbol, timeframe, start_date):
    since = exchange.parse8601(start_date)
    all_candles = []

    while True:
        candles = exchange.fetch_ohlcv(
            symbol,
            timeframe=timeframe,
            since=since,
            limit=1000
        )

        if len(candles) == 0:
            break

        all_candles.extend(candles)

        since = candles[-1][0] + 1

        print(
            f"{symbol} {timeframe}: fetched until "
            f"{datetime.fromtimestamp(candles[-1][0] / 1000, tz=timezone.utc)}"
        )

        time.sleep(exchange.rateLimit / 1000)

        if len(candles) < 1000:
            break

    df = pd.DataFrame(
        all_candles,
        columns=["timestamp", "open", "high", "low", "close", "volume"]
    )

    df["datetime"] = pd.to_datetime(df["timestamp"], unit="ms", utc=True)
    df.set_index("datetime", inplace=True)

    return df[["open", "high", "low", "close", "volume"]]


# =========================================================
# INDICATORS
# =========================================================

def add_indicators(df):
    df = df.copy()

    close = df["close"]
    high = df["high"]
    low = df["low"]

    # EMAs
    df["ema20"] = ta.trend.ema_indicator(close, window=20)
    df["ema50"] = ta.trend.ema_indicator(close, window=50)
    df["ema200"] = ta.trend.ema_indicator(close, window=200)

    # RSI
    df["rsi"] = ta.momentum.RSIIndicator(close, window=14).rsi()

    # MACD
    macd = ta.trend.MACD(close)
    df["macd_hist"] = macd.macd_diff()

    # ATR
    df["atr"] = ta.volatility.AverageTrueRange(
        high=high,
        low=low,
        close=close,
        window=14
    ).average_true_range()

    # Previous breakout high
    df["hh20"] = high.rolling(BREAKOUT_LOOKBACK).max().shift(1)

    return df


# =========================================================
# HTF BIAS
# =========================================================

def htf_bullish(row):
    return (
        row["close"] > row["ema200"] and
        row["ema20"] > row["ema50"]
    )


# =========================================================
# MTF CONFIRMATION
# =========================================================

def mtf_confirmation(row):
    return (
        row["close"] > row["ema200"] and
        row["ema20"] > row["ema50"] and
        row["rsi"] > 50 and
        row["macd_hist"] > 0
    )


# =========================================================
# ENTRY SIGNAL
# =========================================================

def entry_signal(row):
    return (
        row["close"] > row["hh20"] and
        row["rsi"] > 50 and
        row["macd_hist"] > 0
    )


# =========================================================
# BACKTEST
# =========================================================

def run_backtest(symbol):

    print("\n" + "=" * 80)
    print(f"RUNNING BACKTEST FOR {symbol}")
    print("=" * 80)

    df_15m = add_indicators(
        fetch_ohlcv(symbol, ENTRY_TIMEFRAME, START_DATE)
    )

    df_1h = add_indicators(
        fetch_ohlcv(symbol, MID_TIMEFRAME, START_DATE)
    )

    df_4h = add_indicators(
        fetch_ohlcv(symbol, HTF_TIMEFRAME, START_DATE)
    )

    trades = []

    df_15m.dropna(inplace=True)
    df_1h.dropna(inplace=True)
    df_4h.dropna(inplace=True)

    for i in range(1, len(df_15m) - 1):

        current_time = df_15m.index[i]
        row_15m = df_15m.iloc[i]

        # =================================================
        # GET LATEST AVAILABLE 1H AND 4H DATA
        # =================================================

        df_1h_filtered = df_1h[df_1h.index <= current_time]
        df_4h_filtered = df_4h[df_4h.index <= current_time]

        if len(df_1h_filtered) == 0 or len(df_4h_filtered) == 0:
            continue

        row_1h = df_1h_filtered.iloc[-1]
        row_4h = df_4h_filtered.iloc[-1]

        # =================================================
        # HIGHER TIMEFRAME BIAS
        # =================================================

        if not htf_bullish(row_4h):
            continue

        # =================================================
        # MID TIMEFRAME CONFIRMATION
        # =================================================

        if not mtf_confirmation(row_1h):
            continue

        # =================================================
        # ENTRY TIMEFRAME SIGNAL
        # =================================================

        if not entry_signal(row_15m):
            continue

        # =================================================
        # ENTRY EXECUTION
        # =================================================

        next_row = df_15m.iloc[i + 1]

        entry_time = next_row.name

        # Realistic slippage
        entry_price = next_row["open"] * (1 + SLIPPAGE_RATE)

        atr = row_15m["atr"]

        stop_loss = entry_price - atr

        risk_per_unit = entry_price - stop_loss

        if risk_per_unit <= 0:
            continue

        take_profit = entry_price + (risk_per_unit * RR_RATIO)

        # Position sizing using fixed $10 risk
        position_size = RISK_PER_TRADE / risk_per_unit

        entry_value = position_size * entry_price
        entry_fee = entry_value * FEE_RATE

        # =================================================
        # MONITOR TRADE
        # =================================================

        trade_closed = False

        for j in range(i + 1, len(df_15m)):

            future_row = df_15m.iloc[j]

            high = future_row["high"]
            low = future_row["low"]

            exit_reason = None
            exit_price = None

            # Stop Loss hit
            if low <= stop_loss:
                exit_reason = "STOP LOSS"
                exit_price = stop_loss * (1 - SLIPPAGE_RATE)

            # Take Profit hit
            elif high >= take_profit:
                exit_reason = "TAKE PROFIT"
                exit_price = take_profit * (1 - SLIPPAGE_RATE)

            if exit_reason:

                exit_time = future_row.name

                exit_value = position_size * exit_price
                exit_fee = exit_value * FEE_RATE

                gross_pnl = (
                    (exit_price - entry_price)
                    * position_size
                )

                total_fees = entry_fee + exit_fee

                net_pnl = gross_pnl - total_fees

                r_multiple = net_pnl / RISK_PER_TRADE

                holding_period = exit_time - entry_time

                trade = {
                    "symbol": symbol,
                    "entry_time": entry_time,
                    "exit_time": exit_time,
                    "entry_price": entry_price,
                    "exit_price": exit_price,
                    "stop_loss": stop_loss,
                    "take_profit": take_profit,
                    "position_size": position_size,
                    "gross_pnl": gross_pnl,
                    "fees": total_fees,
                    "net_pnl": net_pnl,
                    "r_multiple": r_multiple,
                    "holding_period": holding_period,
                    "exit_reason": exit_reason
                }

                trades.append(trade)

                trade_closed = True
                break

        if not trade_closed:
            continue

    return trades


# =========================================================
# METRICS
# =========================================================

def calculate_metrics(trades):

    if len(trades) == 0:
        return {}

    df = pd.DataFrame(trades)

    wins = df[df["net_pnl"] > 0]
    losses = df[df["net_pnl"] <= 0]

    total_trades = len(df)

    win_rate = (len(wins) / total_trades) * 100

    gross_profit = wins["net_pnl"].sum()
    gross_loss = abs(losses["net_pnl"].sum())

    profit_factor = (
        gross_profit / gross_loss
        if gross_loss != 0
        else np.inf
    )

    expectancy = df["net_pnl"].mean()

    avg_win = wins["net_pnl"].mean() if len(wins) > 0 else 0
    avg_loss = losses["net_pnl"].mean() if len(losses) > 0 else 0

    return {
        "total_trades": total_trades,
        "wins": len(wins),
        "losses": len(losses),
        "win_rate": win_rate,
        "gross_profit": gross_profit,
        "gross_loss": gross_loss,
        "net_pnl": df["net_pnl"].sum(),
        "avg_win": avg_win,
        "avg_loss": avg_loss,
        "expectancy": expectancy,
        "profit_factor": profit_factor,
        "best_trade": df["net_pnl"].max(),
        "worst_trade": df["net_pnl"].min(),
        "avg_r": df["r_multiple"].mean()
    }


# =========================================================
# PRINT TRADE HISTORY
# =========================================================

def print_trades(trades):

    print("\n" + "=" * 120)
    print("TRADE HISTORY")
    print("=" * 120)

    if len(trades) == 0:
        print("No trades found.")
        return

    for idx, trade in enumerate(trades, start=1):

        print(f"\nTRADE #{idx}")
        print("-" * 80)

        print(f"Symbol:          {trade['symbol']}")
        print(f"Entry Time:      {trade['entry_time']}")
        print(f"Exit Time:       {trade['exit_time']}")
        print(f"Holding Period:  {trade['holding_period']}")

        print(f"\nEntry Price:     {trade['entry_price']:.4f}")
        print(f"Stop Loss:       {trade['stop_loss']:.4f}")
        print(f"Take Profit:     {trade['take_profit']:.4f}")
        print(f"Exit Price:      {trade['exit_price']:.4f}")

        print(f"\nPosition Size:   {trade['position_size']:.6f}")

        print(f"\nGross PnL:       ${trade['gross_pnl']:.2f}")
        print(f"Fees:            ${trade['fees']:.2f}")
        print(f"Net PnL:         ${trade['net_pnl']:.2f}")

        print(f"R Multiple:      {trade['r_multiple']:.2f}R")
        print(f"Exit Reason:     {trade['exit_reason']}")


# =========================================================
# PRINT SUMMARY
# =========================================================

def print_summary(metrics):

    print("\n" + "=" * 80)
    print("BACKTEST SUMMARY")
    print("=" * 80)

    if not metrics:
        print("No completed trades.")
        return

    print(f"Total Trades:    {metrics['total_trades']}")
    print(f"Wins:            {metrics['wins']}")
    print(f"Losses:          {metrics['losses']}")
    print(f"Win Rate:        {metrics['win_rate']:.2f}%")

    print(f"\nNet PnL:         ${metrics['net_pnl']:.2f}")
    print(f"Gross Profit:    ${metrics['gross_profit']:.2f}")
    print(f"Gross Loss:      ${metrics['gross_loss']:.2f}")

    print(f"\nAverage Win:     ${metrics['avg_win']:.2f}")
    print(f"Average Loss:    ${metrics['avg_loss']:.2f}")

    print(f"Expectancy:      ${metrics['expectancy']:.2f}")
    print(f"Profit Factor:   {metrics['profit_factor']:.2f}")

    print(f"\nAverage R:       {metrics['avg_r']:.2f}R")
    print(f"Best Trade:      ${metrics['best_trade']:.2f}")
    print(f"Worst Trade:     ${metrics['worst_trade']:.2f}")


# =========================================================
# MAIN
# =========================================================

all_trades = []

for symbol in SYMBOLS:

    trades = run_backtest(symbol)

    print_trades(trades)

    all_trades.extend(trades)

portfolio_metrics = calculate_metrics(all_trades)

print_summary(portfolio_metrics)
